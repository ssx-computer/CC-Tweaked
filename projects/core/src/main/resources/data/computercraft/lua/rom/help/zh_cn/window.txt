blit( text, textColor, backgroundColor )

clearLine()

setCursorBlink( blink )
isColor()
setTextColor( color )
setBackgroundColor( color )
getTextColor()
getBackgroundColor()
getSize()
scroll( n )
setVisible( bVisible )
redraw()
restoreCursor()
getPosition()
reposition( x, y, width, height )
getPaletteColor( color )
setPaletteColor( color, r, g, b )
getLine()
